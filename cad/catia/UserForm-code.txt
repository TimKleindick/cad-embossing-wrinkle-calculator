' Paste into the code window of a manually created UserForm named frmWrinkle.
' This is code, not an importable .frm/.frx or .catvba binary.
Option Explicit

Private Sub UserForm_Initialize()
    Me.Caption = "Faltenrechner - Source Preview - CAD ungetestet"
    txtL1.Text = "40"
    txtL2.Text = "30"
    txtThickness.Text = "2"
    chkThickness.Value = False
    txtResult.Text = "Excel-Standard: 1000 Segmente; Formfit ohne Laengenbedingung."
End Sub

Private Function ShowNumber(ByVal v As Variant, ByVal unit As String) As String
    If IsEmpty(v) Then
        ShowNumber = "entfaellt / nicht angegeben"
    Else
        ShowNumber = Format$(v, "0.000000000") & " " & unit
    End If
End Function

Private Sub cmdCalculate_Click()
    On Error GoTo Failed
    Dim values As Variant, thickness As Variant, names As Variant, units As Variant
    Dim i As Long, output As String
    thickness = Empty
    If chkThickness.Value Then thickness = CDbl(txtThickness.Text)
    values = WrinkleCalculate(CDbl(txtL1.Text), CDbl(txtL2.Text), thickness)
    names = Array("A", "R", "Hoehe", "Winkel", "Gerade je Seite", "Excel-Polygonlaenge", _
        "Analytische Fitlaenge", "Polygon-Laengenabweichung", "Vertikale Formabweichung RMS", _
        "Max. vertikale Formabweichung", "Excel-Kosinusradius", "Korrigierter Kosinusradius", _
        "Kreisbogen-Dehnungsabschaetzung", "Excel-Dehnungsabschaetzung", "Korrigierte Kosinusabschaetzung")
    units = Array("mm", "mm", "mm", "deg", "mm", "mm", "mm", "%", "mm", "mm", "mm", "mm", "%", "%", "%")
    output = "SOURCE PREVIEW - CAD-Laufzeit/Import ungetestet" & vbCrLf
    output = output & "Excel: 1000 Segmente, Formfit ohne Laengenbedingung" & vbCrLf
    For i = 0 To 14
        output = output & names(i) & ": " & ShowNumber(values(i), units(i)) & vbCrLf
    Next
    If Abs(values(7)) >= 0.1 Then output = output & "Laengenziel <0.1% NICHT erreicht." & vbCrLf
    output = output & "Dehnung: stark vereinfachtes t/(2r)-Modell; keine Materialfreigabe oder Risssicherheitsaussage."
    txtResult.Text = output
    Exit Sub
Failed:
    txtResult.Text = "Fehler: " & Err.Description & vbCrLf & "Kein Ergebnis freigegeben."
End Sub

Private Sub cmdClose_Click()
    Unload Me
End Sub
