' Paste into the code window of a manually created UserForm named frmWrinkle.
' This is code, not an importable .frm/.frx or .catvba binary.
Option Explicit

Private Sub UserForm_Initialize()
    Me.Caption = "Wrinkle Geometry - Source Preview"
    txtL1.Text = "40"
    txtL2.Text = "30"
    txtThickness.Text = "2"
    chkThickness.Value = False
    txtResult.Text = "Excel method with 1,000 segments and no fitted-length constraint."
End Sub

Private Function ShowNumber(ByVal v As Variant, ByVal unit As String) As String
    If IsEmpty(v) Then
        ShowNumber = "Not applicable or not provided"
    Else
        ShowNumber = Format$(v, "0.000000000") & " " & unit
    End If
End Function

Private Sub cmdCalculate_Click()
    On Error GoTo Failed
    Dim values As Variant, thickness As Variant, names As Variant, units As Variant
    Dim i As Long, output As String
    thickness = Empty
    If chkThickness.Value Then thickness = CDbl(txtThickness.Text)
    values = WrinkleCalculate(CDbl(txtL1.Text), CDbl(txtL2.Text), thickness)
    names = Array("A", "R", "Height h", "Flank angle", "Straight length per side", "Fitted polyline length", _
        "Analytic fitted length", "Polyline length deviation", "RMS vertical profile deviation", _
        "Maximum vertical deviation at sample points", "Cosine radius using the Excel formula", "Corrected cosine radius", _
        "Circular arc strain estimate", "Cosine strain using the Excel radius", "Cosine strain using the corrected radius")
    units = Array("mm", "mm", "mm", "deg", "mm", "mm", "mm", "%", "mm", "mm", "mm", "mm", "%", "%", "%")
    output = "SOURCE PREVIEW - CAD runtime and import untested" & vbCrLf
    output = output & "Excel method with 1,000 segments and no fitted-length constraint" & vbCrLf
    For i = 0 To 14
        output = output & names(i) & ": " & ShowNumber(values(i), units(i)) & vbCrLf
    Next
    If Abs(values(7)) >= 0.1 Then output = output & "Length deviation reaches or exceeds 0.1%." & vbCrLf
    output = output & "Simplified t/(2r) estimate. Material approval and crack safety are outside its scope."
    txtResult.Text = output
    Exit Sub
Failed:
    txtResult.Text = "Error: " & Err.Description & vbCrLf & "No result."
End Sub

Private Sub cmdClose_Click()
    Unload Me
End Sub
